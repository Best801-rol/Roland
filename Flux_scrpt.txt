import pandas as pd

1. Préparer les données
df["Date"] = pd.to_datetime(df["Date"], errors="coerce")
df["Montant"] = pd.to_numeric(df["Montant"], errors="coerce")

2. Agréger les opérations par FCP et date
resultat = (
df.groupby(["FCP", "Date", "TypeOperation"], dropna=False)
.agg(
nombre_operations=("TypeOperation", "size"),
montant_total=("Montant", "sum")
)
.reset_index()
)

3. Séparer les souscriptions et les rachats
resultat = resultat.pivot_table(
index=["FCP", "Date"],
columns="TypeOperation",
values=["nombre_operations", "montant_total"],
aggfunc="sum",
fill_value=0
)

4. Aplatir les colonnes
resultat.columns = [
f"{indicateur}_{type_op}"
for indicateur, type_op in resultat.columns
]
resultat = resultat.reset_index()

5. Créer les quatre colonnes finales
colonnes = {
"nombre_operations_Souscription": "nombre_totaloperation_souscription",
"montant_total_Souscription": "montant_total_souscription",
"nombre_operations_Rachat": "nombre_totaloperation_rachat",
"montant_total_Rachat": "montant_total_rachat"
}

resultat = resultat.rename(columns=colonnes)

Garantir la présence des quatre colonnes
for col in colonnes.values():
if col not in resultat.columns:
resultat[col] = 0

print(resultat)
