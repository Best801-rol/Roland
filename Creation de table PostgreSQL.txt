CREATE SCHEMA IF NOT EXISTS core;
CREATE SCHEMA IF NOT EXISTS reporting;

CREATE TABLE core.clients (
client_id TEXT PRIMARY KEY,
nom TEXT,
ville TEXT
);

CREATE TABLE core.contrats (
contrat_id TEXT PRIMARY KEY,
client_id TEXT NOT NULL REFERENCES core.clients(client_id),
date_contrat DATE,
montant_total NUMERIC(14, 2) NOT NULL,
statut TEXT NOT NULL
);

CREATE TABLE core.echeances (
paiement_id TEXT PRIMARY KEY,
contrat_id TEXT NOT NULL REFERENCES core.contrats(contrat_id),
date_echeance DATE NOT NULL,
montant_du NUMERIC(14, 2) NOT NULL,
montant_paye NUMERIC(14, 2) NOT NULL DEFAULT 0
);

INSERT INTO staging.clients (client_id, nom, ville)
SELECT DISTINCT
NULLIF(TRIM(client_id), ''),
NULLIF(TRIM(nom), ''),
NULLIF(INITCAP(TRIM(ville)), '')
FROM raw.clients
WHERE NULLIF(TRIM(client_id), '') IS NOT NULL;

Cette requête :
Supprime les espaces inutiles.
Remplace les chaînes vides par NULL.
Harmonise la casse des villes et supprime les doublons strictement identiques.
Écarte les lignes sans identifiant client.
